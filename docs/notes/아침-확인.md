   ```
   sudo nmcli con add type ethernet ifname eth0 con-name haedam-wan ipv4.method manual ipv4.addresses "203.0.113.10/24,203.0.113.11/24,203.0.113.12/24,203.0.113.21/24,203.0.113.22/24,203.0.113.41/24,203.0.113.100/24" ipv4.gateway 203.0.113.1 && sudo nmcli con up haedam-wan
   ```
